#include <iostream>
#include <string>
#include <vector>

using namespace std;

int main() {

    string sentence;
    vector <string> strVec(100);
    while (getline(cin, sentence)){
        strVec.push_back(sentence);
    }

    int maxLen = strVec[0].length();
    for (auto i : strVec){
        if(i.length() >= maxLen)
            maxLen = i.length();
    }

    for (int j = 0; j < strVec.size(); j++){
        if (strVec[j].length() < 100){
            for (int a = 0; a < 100 - strVec[j].length(); a++){
                strVec[j] += " ";
            }
        }
    }
    
    for (int k = 0; k < maxLen; k++){
        for (int l = strVec.size() - 1; l >= 0; l--){
                cout << strVec[l][k];
        }
        cout << endl;
    }

    return 0;
}
