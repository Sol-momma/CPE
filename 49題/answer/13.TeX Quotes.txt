#include <iostream>
#include <string>

using namespace std;

int main() {

    string sentence;
    const string leftQuote = "``";
    const string rightQuote = "''";
    int quote = 0;

    while (getline(cin, sentence)){
        for (int i = 0; i < sentence.length(); i++){
            if (sentence[i] == '"'){
                quote++;
                if (quote % 2 != 0)
                    cout << leftQuote;
                else
                    cout << rightQuote;
            }
            else
                cout << sentence[i];
        }
        cout << endl;
    }

    return 0;
}
