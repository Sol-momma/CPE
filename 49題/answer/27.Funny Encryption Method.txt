27.Funny Encryption Method

#include <iostream>
#include <string>
#include <cmath>

using namespace std;

int binary(int num){
	int ans = 0;
	while(num > 0){
			if(num % 2 != 0){
				ans++;
			}
			num /= 2;
	}
	return ans;
}

int main(){
	int test;
	cin >> test;
	string num;
	for(int i = 0; i < test; i++){
		cin >> num;
		int decimal = stoi(num);
		int ans1 = binary(decimal);
		
		int hexa = 0;
		int temp;
		for(int j = num.size() - 1; j >= 0; j--){
			temp = num[j] - '0';
			hexa += temp * pow(16, num.size() - 1 - j);
		}
		int ans2 = binary(hexa);
		
		cout << ans1 << " " << ans2 << endl;
	}

	return 0;
}
