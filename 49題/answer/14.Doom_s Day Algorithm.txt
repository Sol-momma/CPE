日付に+5すると余り0のとき日曜日

#include <bits/stdc++.h>

using namespace std;

int main() {
    int kase;
    cin >> kase;
    string Day[]={"Sunday","Monday","Tuesday","Wednesday","Thursday","Friday","Saturday"};
    int Month[]={0,31,28,31,30,31,30,31,31,30,31,30,31};
    while (kase--) {
        int m, d;
        cin >> m >> d;
        int days=0; //今年に入って何日目か
        for(int i=0;i<m;i++)
            days+=Month[i]; //前月までの日付の合計
        int w=(days+d+5)%7; //+今月の日付+5
//問題文より、1/9が日曜日で+5=14 14%7=0 余りが0のとき日曜日
        cout<<Day[w]<<endl;
    }
    return 0;
}
