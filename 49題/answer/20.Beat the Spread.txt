20.Beat the Spread

#include <iostream>

using namespace std;

int main(){
	int test;
	cin >> test;
	
	int sum, diff;
	int big, small;
	for(int i = 0; i < test; i++){
		cin >> sum >> diff;
		
		if(diff > sum - 1){
			cout << "impossible" << endl;
		}
		else{
			big = sum + diff;
			if(big % 2 != 0){
				cout << "impossible" << endl;
			}
			else{
				big /= 2;
				small = sum - big;
				cout << big << " " << small << endl;
			}
		}
	}
	return 0;
}
