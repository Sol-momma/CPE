#include <iostream>

using namespace std;

int main(){
	int test;
	cin >> test;
	
	int day, party, hartal;
	int start = 1;
	for(int i = 0; i < test; i++){
		cin >> day >> party;
		int arr[day + 1] = {0};
		
		for(int j = 0; j < party; j++){
			cin >> hartal;
			for(int k = hartal; k <= day; k += hartal){
				if(k % 7 != 6 && k % 7 != 0){
					if(arr[k] == 0){
						arr[k] = 1;
					}
				}
			}
		}
		
		int count = 0;
		for(int j = 0; j <= day; j++){
			if(arr[j] != 0){
				count++;
			}
		}
		cout << count << endl;
	}
	
	return 0;
}
