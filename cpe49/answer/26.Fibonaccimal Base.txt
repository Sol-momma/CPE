#include <iostream>
#include <vector>
#include <string>

using namespace std;

int main(){
	int test;
	cin >> test;
	
	for(int i = 0; i < test; i++){
		int integ;
		cin >> integ;
		int output = integ;
		
		vector<int> fibo {1, 2};
		int input = 0;
		int index = 1;
		while(fibo.back() <= integ){
			fibo.push_back(fibo[index - 1] + fibo[index]);
			index++;
		}
		
		if(integ != fibo.back()){
			index--;
		}
		
		integ -= fibo[index];
		index--;
		string ans = "1";
		while(index >= 0){
			if(ans.back() == '1'){
				ans.push_back('0');
				index--;
			}
			else if(fibo[index] <= integ){
				ans.push_back('1');
				integ -= fibo[index];
				index--;
			}
			else{
				ans.push_back('0');
				index--;
			}
		}
		cout << output << " = " << ans << " (fib)" << endl;
	}
	return 0;
}
