28.Parity

#include <iostream>
#include <string>
#include <algorithm>

using namespace std;

int main(){
	
	int num;
	cin >> num;
	while(num != 0){
		int parity = 0;
		string binary = "";
		while(num > 0){
			if(num % 2 != 0){
				parity++;
				binary.push_back('1');
			}
			else{
				binary.push_back('0');
			}
			num /= 2;
		}
		
		reverse(binary.begin(), binary.end());
		cout << "The parity of " << binary << " is " << parity << " (mod 2)." << endl;
		
		cin >> num;
	}
	
	return 0;
}
