#include <iostream>
#include <cmath> //absを使う

using namespace std;

int main() {
    long long firstNum, secondNum;
    //unsignd int = 0 ~ 2^32-1 : long long = -2^63 ~ -2^63-1

    while (cin >> firstNum >> secondNum){
        long long difference = abs(firstNum - secondNum);
        cout << difference << endl;
    }

    return 0;
}


この問題では2^32までの範囲
