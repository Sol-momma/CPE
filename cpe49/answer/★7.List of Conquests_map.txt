＊mapを使う！！


#include <iostream>
#include <string>
#include <map>

using namespace std;

int main() {
    int testCase;
    cin >> testCase;
    string countryName;
    map <string, int> numOfPeople; //map <kayの型, valueの型> mapの名前
//map名.emplace(kay, value)で要素の追加　勝手にソートされる
　　
    for(int i = 0; i < testCase; i++){
	cin >> countryName; //まず国名だけ取得してmapに入れる
        numOfPeople[countryName]++;
//kayが同じところのvalueに+1 //[]を使うと、もし同じのがなかったら勝手に新しいkayが足される //勝手に昇順になる
        getline(cin, countryname);
//残りの人名は何単語かわからないから、cinで一行すべてをstring型に保存して、使わないからすぐに次のループで国名で上書き
//getline()は改行文字\n以外のすべてを読み取る
    }

    for (auto itr = numOfPeople.begin(); itr != numOfPeople.end(); itr++){ //イテレータ：要素に順番にアクセスする .bigin() から .end() までループ処理
        cout << itr->first << " " << itr->second << endl;
    }

    return 0;
}
