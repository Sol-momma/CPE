#include <iostream>
#include <string>

using namespace std;

int main() {
    char firstKeybord[13] = {'`', '1', '2', '3', '4', '5', '6', '7', '8', '9', '0', '-', '='};
    char secondKeybord[12] = {'q', 'w', 'e', 'r', 't', 'y', 'u', 'i', 'o', 'p', '[', ']'};
    char thirdKeybord[10] = {'a', 's', 'd', 'f', 'g', 'h', 'j', 'k', 'l', ';'};
    char forthKeybord[10] = {'z', 'x', 'c', 'v', 'b', 'n', 'm', ',', '.', '/'};

    string encodedMessage;
    getline(cin, encodedMessage);

    char currentChar;

    for(int i = 0; i < encodedMessage.length(); i++){
        bool isInArray = false;
        currentChar = tolower(encodedMessage[i]);
        if (currentChar == ' '){
            cout << " ";
            continue;
        }
        for(int j = 0; j < 13; j++){
            if(currentChar == firstKeybord[j]){
                cout << firstKeybord[j - 2];
                isInArray = true;
            }
        }
        if (!isInArray){
            isInArray = false;
            for(int j = 0; j < 12; j++){
                if(currentChar == secondKeybord[j]){
                    cout << secondKeybord[j - 2];
                    isInArray = true;
                }
            }
        }
        else
            continue;

        if (!isInArray){
            isInArray = false;
            for(int j = 0; j < 10; j++){
                if(currentChar == thirdKeybord[j]){
                    cout << thirdKeybord[j - 2];
                    isInArray = true;
                }
            }
        }
        else
            continue;

        if (!isInArray){
            isInArray = false;
            for(int j = 0; j < 10; j++){
                if(currentChar == forthKeybord[j]){
                    cout << forthKeybord[j - 2];
                    isInArray = true;
                }
            }
        }
    }

    return 0;
}
