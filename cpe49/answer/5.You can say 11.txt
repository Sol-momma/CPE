#include <iostream>

using namespace std;

int main() {
    unsigned long long positiveNum;

    while (true){
        cin >> positiveNum;

        if (positiveNum == 0)
            break;
        else if (positiveNum % 11 == 0)
            cout << positiveNum << " is a multiple of 11." << endl;
        else
            cout << positiveNum << " is not a multiple of 11." << endl;
    }

    return 0;
}

#include <iostream>
#include <string>

using namespace std;

int main() {
    string positiveNum;

    while (true){
        cin >> positiveNum;
        int oddDigit = 0, evenDigit = 0;

        if (positiveNum == "0")　//positiveNum[0] == '0'にしたらダメだった
            break;

        for (int i = 0; i < positiveNum.length(); i += 2){
            oddDigit += positiveNum[i] - '0';
        }
        for (int i = 1; i < positiveNum.length(); i += 2){
            evenDigit += positiveNum[i] - '0';
        }

        int diff = abs(oddDigit - evenDigit);

        if (diff % 11 == 0)
            cout << positiveNum << " is a multiple of 11." << endl;
        else
            cout << positiveNum << " is not a multiple of 11." << endl;
    }

    return 0;
}
